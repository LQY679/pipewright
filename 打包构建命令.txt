Windows 打包构建说明

# 一键构建(自动生成版本元数据 ldflags,无需手动拼 $LDFLAGS)
# 用法见 build.ps1 顶部注释;前端产物 web/dist 已存在时会自动复用。

# 交叉编译为 linux/arm64 (静态链接, 无外部依赖, 产物 -> dist/linux-arm64/pipewright)
.\build.ps1 linux-arm64

# 交叉编译为 linux/x86_64 (产物 -> dist/linux-x86_64/pipewright)
.\build.ps1 linux-x86_64

# 编译为 windows/x86_64 (产物 -> dist/windows-x86_64/pipewright.exe)
.\build.ps1 windows-x86_64

# 本机平台编译 (产物 -> dist/local/pipewright[.exe])
.\build.ps1 local

# 改动过 web/ 源码后,需要强制重建前端再打包:
.\build.ps1 linux-arm64 -RebuildFrontend

# 查看二进制注入的版本信息(验证 ldflags 注入成功):
.\dist\linux-arm64\pipewright.exe --version   # Windows 上直接运行
dist/linux-arm64/pipewright --version          # Linux 目标(需要对应平台环境)

# 手动构建(可选,不推荐):等价于下面三行
$env:GOOS="linux"; $env:GOARCH="arm64"; $env:CGO_ENABLED="0"
go build -ldflags $LDFLAGS -o dist/linux-arm64/pipewright ./cmd/pipewright
Remove-Item Env:GOOS, Env:GOARCH, Env:CGO_ENABLED -ErrorAction SilentlyContinue
